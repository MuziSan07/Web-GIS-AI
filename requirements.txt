# Deployment requirements for the hosted demo.
#
# The engine's own dependencies live in engine/pyproject.toml. This file is what
# a host such as Streamlit Community Cloud installs, and it deliberately differs
# from that file in two ways.
#
# Versions are chosen so every dependency installs from a wheel on Python 3.12.
# Building rasterio or shapely from source on a shared container is slow enough
# to time out a deploy, and the wheel coverage is genuinely uneven:
#   rasterio 1.5 publishes no wheel for 3.11
#   ezdxf 1.4 publishes none for 3.14
#   reportlab 5 dropped its pure python wheel; 4.x still has one
# Python 3.12 is the version every one of them covers. Set it in the host's
# advanced settings.
streamlit>=1.40

numpy>=2.0
scipy>=1.11
shapely>=2.0
pyproj>=3.6
geopandas>=1.0
rasterio>=1.5
pyyaml>=6.0
pydantic>=2.0

# Export writers
ezdxf>=1.3,<1.5
reportlab>=4.4,<5
ifcopenshell>=0.8

# The engine itself. Installed from the repository, not editable: an editable
# install needs a writable checkout, which a build container does not promise.
./engine
